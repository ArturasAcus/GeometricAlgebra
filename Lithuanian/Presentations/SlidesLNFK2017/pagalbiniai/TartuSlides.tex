\section[Standard quantum mechanics;]{Standard quantum mechanics}
%\subsection{Standard formulation}
\begin{frame}
\frametitle<presentation>{Standard quantum mechanics approach}
\begin{block}{Simplest example: electron in a monolayer graphene}
In the Hilbert space formalism, the \alert{massless Dirac
Hamiltonian} in the vicinity of critical $K$  point of graphene
reduces to simple form
\begin{equation}\label{Hgmat}
\hat{H}_{MG}= \hbar v_F\left[\begin{array}{cc}
 0& k_x-\ii k_y\\
 k_x+\ii k_y&0
\end{array}\right],
\end{equation}
where $v_F\approx 10^{6}$~m/s is the Fermi velocity. The wave
vectors $\mathbf{k}=(k_x,k_y)$ lies in the graphene plane and is
measured with respect to valley minimum. In the following the
natural units $\hbar=1$ along with $v_F=1$ are used.
\end{block}
\end{frame}

%--7--------------------------------

\begin{frame}
\frametitle<presentation>{Standard quantum mechanics approach}
\begin{columns}[t]
\begin{column}{0.5\textwidth}
     \begin{block}{Eigenenergies}
The eigenenergies of the Hamiltonian~\eqref{Hgmat} are
\begin{equation}\label{dispK2}
E_{\pm}=\pm\sqrt{k_x^2+k_y^2}=\pm|\mathbf{k}|,
\end{equation}
where plus/minus sign  corresponds to conduction/valence bands.
Since $E\propto |\mathbf{k}|$, the spectrum is ultrarelativistic.
\end{block}
\end{column}
\begin{column}{0.5\textwidth}
     \begin{block}{Spectrum of monolayer graphene}
\includegraphics[width=0.8\textwidth,height=0.65\textwidth]{figures//graphenespec3.jpg}

{\color{black}\small  CB/VB means conduction/valence band.
Left/right arrows show pseudospin direction.}
\end{block}
\end{column}
\end{columns}
\end{frame}

\section[{\it Cl}$_{3,0}$ case;]{Geometric algebra approach: Cl$_{3,0}$ case}
\subsection{Example of monolayer graphene}
\begin{frame}
\vspace{-20pt}
\frametitle<presentation>{\textit{Cl}$_{3,0}$ algebra approach}
\begin{columns}[t]
\begin{column}{0.47\textwidth}
     \begin{block}{\textit{Cl}$_{3,0}$ algebra}
\small Minimal algebra to describe the monolayer Hamiltonian is \textit{Cl}$_{3,0}$.
\includegraphics[width=0.99\textwidth]{figures//graphenePlane.jpg}
\newline
\small Base vectors $\e{1}^2=\e{2}^2=\e{3}^2=1$ play the role of space coordinates.  The orthogonal oriented planes are bivectors $\Ie{1}=\e{2}\e{3}$, $\Ie{2}=\e{3}\e{1}$,
and $\Ie{3}=\e{1}\e{2}$. With scalar and pseudoscalar
$I=\e{1}\e{2}\e{3}$ included, there are  8 basis elements.
\end{block}
\end{column}
\begin{column}{0.57\textwidth}
 \begin{block}{Replacement rules}
\small Transform the Hamiltonian~\eqref{Hgmat} to
\textit{Cl}$_{3,0}$ using mapping rules from
Hilbert to Clifford space~\cite{Doran03}\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}\begin{align}\label{repRules}
|\psi\rangle &\Leftrightarrow\psi=a_0+a_1\Ie{1}+a_2\Ie{2}+a_3\Ie{3},\notag\\
\lambda|\psi\rangle &\Leftrightarrow\lambda\psi,\notag \\
\hat{\sigma}_i|\psi\rangle &\Leftrightarrow\e{i}\psi\e{3}.
\end{align}
$\lambda$ is the scalar and $\hat{\sigma}_i$ are Pauli
matrices. Then~\eqref{Hgmat}, $\hat{H}_{MG}|\psi\rangle$, maps to the following simple GA
function of spinor $\psi$
\begin{equation}\label{HkinGA}
\sH_{MG}(\psi)=\mathbf{k}\psi\e{3}.
\end{equation}
The wave vector $\mathbf{k}=k_x\e{1}+k_y\e{2}$ lies in the
graphene plane.  $\e{3}$ represents the pseudospin quantization
axis, which is perpendicular to graphene plane.
\end{block}
\end{column}
\end{columns}
\end{frame}

%--10-------------------------------------------------

\begin{frame}
\vspace{-20pt}
\frametitle<presentation>{\textit{Cl}$_{3,0}$ algebra approach}
\begin{columns}[t]
\begin{column}{0.47\textwidth}
     \begin{block}{Eigenvalue equation}
\small From \eqref{HkinGA} eigenvalue equation for monolayer graphene follows
\begin{equation}
\bk\psi\e{3}=\psi E,
\end{equation}
which can be easily rewritten as an equation for rotor
\begin{equation}\label{rotMG}
\frac{\psi\e{3}\tilde{\psi}}{\psi\tilde{\psi}}=\frac{\bk}{|\bk|^2}E,
\end{equation}
where $|\bk|\equiv k=\sqrt{k_x^2+k_y^2}$ is the magnitude of the
wave vector. Since $\e{3}$ is a unit vector, after
rotation by $\psi$ we shall obtain the unit vector on the right
hand side too.
\end{block}
\end{column}
\begin{column}{0.57\textwidth}
 \begin{block}{Eigenvalues}
\small This means that the square of right hand side
of~\eqref{rotMG} must be $1$, $(E/k)^2=1$, from which follows
2D ultrarelativistic dispersion law $E_{\pm}=\pm
k=\pm\sqrt{k_x^2+k_y^2}$.
\includegraphics[width=0.99\textwidth]{figures//QWaxes.jpg}

Electron with velocity $\mathbf{v}$ and
coordinate system for graphene. The planes
represent graphene layers. $\mathbf{k}$ is electron wave vector.
$\theta=\pi/2$, $\cos\varphi=k_1/|\bk|$,
$\sin\varphi=k_2/|\bk|$.
\end{block}
\end{column}
\end{columns}
\end{frame}
%--11---------------------------
\begin{frame}
\vspace{-20pt}
\frametitle<presentation>{\textit{Cl}$_{3,0}$ algebra approach}
\begin{columns}[t]
\begin{column}{0.99\textwidth}
\begin{block}{Eigenmultivectors of monolayer graphene}
The corresponding eigenmultivectors can be deduced from the same
equation~\eqref{rotMG} after insertion of eigenvalue  $E_{\pm}$
back and requiring the \alert{rotor-spinor $\psi$} to satisfy
$\psi\tilde{\psi}=1$. Then, the equation~\eqref{rotMG} reduces to
\begin{equation}\label{rot2}
\psi_{\pm}\e{3}\tilde{\psi}_{\pm}=\pm\hat{\mathbf{k}},
\end{equation}
where $\hat{\bk}=\mathbf{k}/|\mathbf{k}|$. Thus, according to the
geometric algebra the eigenspinors $\psi_{\pm}$ of a monolayer
graphene are rotors that bring  the unit vector $\e{3}$ that is
parallel  to quantization axis to new direction $\pm\hat{\bk}$,
which is either parallel or antiparallel to the electron wave
vector $\bk$.
\end{block}
\end{column}
\end{columns}
\begin{columns}[t]
\begin{column}{0.35\textwidth}
\vbox{\includegraphics[width=1.0\textwidth]{figures//QWaxes.jpg}}
\end{column}
\begin{column}{0.65\textwidth}
\vspace{-50pt}

\alert{No such clear-cut geometrical interpretation exists
for eigenspinors calculated in the Hilbert space.}
\end{column}
\end{columns}
\end{frame}

%--12---------------------------
\begin{frame}
\vspace{-20pt}
\frametitle<presentation>{\textit{Cl}$_{3,0}$ algebra approach}
\begin{columns}[t]
\begin{column}{0.99\textwidth}
\begin{block}{Eigenmultivectors of monolayer graphene}
The solution of
\eqref{rot2} can be written automatically as a rotor in the unit
bivector $\hat{\mathbf{k}}\w\e{3}$ plane,
{\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\begin{equation}\label{psiMG}
\psi_{\pm}=\ee^{\pm\hat{\mathbf{k}}\W\e{3}\,\pi/4}
=\frac{1}{\sqrt{2}}\Big(1\pm\hat{\mathbf{k}}\w\e{3}\Big).
\end{equation}}
Since $\hat{\mathbf{k}}$ and $\e{3}$ are mutually perpendicular
the outer product may be replaced by geometric one,
$\hat{\mathbf{k}}\w\e{3}=\hat{\mathbf{k}}\e{3}$. In the expanded
coordinate form the eigenspinors  are
\begin{equation}
\psi_{\pm}=\frac{1}{\sqrt{2}}\Big(1\pm\sin\varphi\Ie{1}\mp\cos\varphi\Ie{2}\Big).
\end{equation}
The eigenspinors~\eqref{psiMG} are written
in a coordinate-free way. The appearance of $\e{3}$
in~\eqref{psiMG} is indispensable since the quantum mechanics
requires to indicate the direction of the pseudospin (or spin)
quantization axis which in our case coincides with $\e{3}$.
\end{block}
\end{column}
\end{columns}
\end{frame}

\end{equation}
\end{block}
\end{column}

\subsection{The algorithm}

\begin{frame}
\vspace{-20pt}
\frametitle<presentation>{The algorithm}
\begin{columns}[t]
\begin{column}{0.99\textwidth}
\begin{block}{Important steps}
\begin{itemize}
\item <1->Using replacement rules rewrite the hamiltonian in terms
of geometric algebra

\item <2->Put obtained equation into rotor equation form

\item <3->Solve algebraic equation for eigenenergy
\item <4->Substitute eigenenergy into rotor equation and solve for eigenmultivector.
\end{itemize}
\end{block}
Test how the algorithm works for other \textit{Cl}$_{3,0}$ algebra problem, i.~e. for
electron in quantum well.
\end{column}
\end{columns}
\end{frame}

%--18-------------------------
\subsection{One more example: electron in quantum well}
\begin{frame}
\vspace{-20pt}
\frametitle<presentation>{\textit{Cl}$_{3,0}$ algebra approach}
\begin{columns}[t]
\begin{column}{0.99\textwidth}
\begin{block}{Electron in quantum well}
Quantum well Hamiltonian with spin-orbit interaction
$\alpha_{\rm{R}}$ {\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\begin{equation}\label{ham2D}
\hat{H}_{QW}=\frac{\hbar^2}{2m^*}(k_x^2+k_y^2)\hat{1}+\alpha_{\rm{R}}(k_y\hat{\sigma}_x-k_x\hat{\sigma}_y)\,.
\end{equation}}
Hamiltonian mapped into \textit{Cl}$_{3,0}$ algebra
{\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\begin{equation}\label{HQW}
\sH_{QW}(\psi)=\frac{1}{2}(k_x^2+k_y^2)\psi+\alpha_{\rm{R}}(k_y\e{1}-k_x\e{2})\psi\e{3}=
\frac{\bk^2}{2}+\alpha_{\rm{R}}\e{12}\bk\psi\e{3}.
\end{equation}}
Multivector equation, where $E$ is the eigenenergy:
{\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\begin{equation}\label{eigQW}
\frac{\bk^2}{2}\psi+\alpha_{\rm{R}}\e{12}\bk\psi\e{3}=E\psi.
\end{equation}}
The equation rearranged into rotor form
\begin{equation}\label{roteq12A}
\psi\e{3}\tilde{\psi}=\frac{k^2/2-E}{\alpha_R k^2}\bk\e{12}\,,
\end{equation}
where $\bk\e{12}=\bk\D\e{12}$.
\end{block}
\end{column}
\end{columns}
\end{frame}

%--19---------------------

\begin{frame}
\vspace{-20pt}
\frametitle<presentation>{\textit{Cl}$_{3,0}$ algebra approach}
\begin{columns}[t]
\begin{column}{0.99\textwidth}
\begin{block}{Electron in quantum well}
Spectrum
{\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\begin{equation}E_{\pm}=\frac{k^2}{2}\pm k\alpha_R\,.
\end{equation}}
Insertion of eigenenergies
$E_{\pm}$  back into \eqref{roteq12A} gives the rotor equation
{\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\begin{equation}\label{roteq12B}
\psi_{\pm}\e{3}\tilde{\psi}_{\pm}=\pm(k_y\e{1}-k_x\e{2})/k\equiv\hat{\mathbf{a}}_{\pm}.
\end{equation}}
Since $\e{3}$ is perpendicular to quantum well plane and
$\hat{\mathbf{a}}_{\pm}$ lies in the quantum well plane the
rotation angle $\theta=\pm\pi/2$. Therefore, the solution of
equation~\eqref{roteq12B} is the rotor
{\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\begin{equation}
\psi_{\pm}=\exp\Big(\pm\frac{\e{3}\w\hat{\mathbf{a}}_{\pm}}{|\e{3}\w\hat{\mathbf{a}}_{\pm}|}\frac{\pi}{4}\Big).
\end{equation}}
In the expanded coordinate form the eigenspinors-rotors are
{\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\begin{equation}\label{psi12}
\psi_{\pm}=(1\mp\cos\varphi\,\e{23}\mp\sin\varphi\,\e{31})/\sqrt{2},
\end{equation}}
where $\cos\varphi=k_x/k$ and $\sin\varphi=k_y/k$.
\end{block}
\end{column}
\end{columns}
\end{frame}



\section[{\it Cl}$_{3,1}$ case;]{Geometric algebra approach: Cl$_{3,1}$ case}


%--21-------------------------
\subsection{A pair of coupled two-level atoms}
\begin{frame}
\vspace{-20pt}
\frametitle<presentation>{\textit{Cl}$_{3,1}$ algebra case}
\begin{columns}[t]
\begin{column}{0.99\textwidth}
\begin{block}{A pair of coupled two-level atoms}
Two-level \alert{single atom} standard Hamiltonian  and its dipole
momentum operator {\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\begin{equation}
\hat{H}_1=\frac{\hbar\omega}{2}\hat{\sigma}_z ,\qquad \hat{\mathbf{d}}_1= \boldsymbol{\mu}\hat{\sigma}_x,
\end{equation}}
where $\hbar\omega$ is the energy difference between the energy
levels and $\boldsymbol{\mu}$ is the coupling constant.
After mapping to the
\textit{Cl}$_{3,0}$ the Hamiltonian and dipole moment become
{\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\begin{equation}
\begin{split}
&\sH_1(\psi)=\frac{\hbar\omega}{2}\e{3}\psi\e{3},\qquad \mathbf{d}_1(\psi)=\boldsymbol{\mu}\e{1}\psi\e{3}.
\end{split}
\end{equation}
} Now take \alert{a pair of two-level atoms} coupled to each
other through dipole-dipole coupling which is proportional to
$\Gamma=\boldsymbol{\mu}^{(1)}\d\boldsymbol{\mu}^{(2)}$. This
interaction may produce entanglement!
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{block}{\textit{Cl}$_{3,1}$ matrix representation}
Basis vectors $\e{i}$ satisfy $\e{1}^2=\e{2}^2=\e{3}^2=1$ and
$\e{4}^2=-1$. Complex $4\times 4$ matrix representation of $\e{i}$
used in decomposition the Hamiltonian, pseudospin and velocity
matrices
\begin{equation*}
\hat{e}_{1}=\left[
\begin{array}{cc}
0&\hat{1}\\
\hat{1}&0\\
\end{array}
\right], \
 \hat{e}_{2}=\ii\left[
\begin{array}{cccc}
0&-\hat{1}\\
\hat{1}&0\\
\end{array}
\right],
\end{equation*}
%
\begin{equation*}
\hat{e}_{3}=\left[
\begin{array}{cc}
\hat{\sigma}_y&0\\
0&-\hat{\sigma}_y\\
\end{array}
\right], \
 \hat{e}_{4}=\ii\left[
\begin{array}{cccc}
\hat{\sigma}_z&0\\
0&-\hat{\sigma}_z\\
\end{array}
\right],
\end{equation*}
where $\ii=\sqrt{-1}\,$. $\hat{1}$ is $2\times 2$ unit matrix and
$\hat{\sigma}_x$, $\hat{\sigma}_y$, $\hat{\sigma}_z$  are Pauli
matrices.
\end{block}
\end{column}
\begin{column}{0.41\textwidth}
 \begin{block}{Replacement rules}
\small The following mapping is assumed between the complex spinor
$|\psi\rangle$ in a form of column and GA spinor $\psi$,
\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\begin{equation}\label{psipsi}
\begin{split}
 |\psi\rangle&=\left[
\begin{array}{c}
\ \,a_0+\ii a_3\\
- b_3+\ii b_0\\
-b_2-\ii b_1\\
-a_1+\ii a_2
\end{array}\right]\\
&\Updownarrow\\
\psi&=a_0+a_1\e{23}
-a_2\e{31}\\
&+a_3\e{12}-b_0I-b_1\e{14}\\
&+b_2\e{24}+b_3\e{34},
\end{split}
\end{equation}
where $a_i$'s and $b_i$'s are scalars.
\end{block}
\end{column}
\end{columns}

\begin{block}{Replacement rules (continued)}
The knowledge of matrix representation of basis elements
and spinor mapping rule~\eqref{psipsi} allow to construct
the main replacement rules between the action of
matrices on column vectors  and \textit{Cl}$_{3,1}$ multivector
functions~\cite{Dargys13bI,Dargys13bII},
\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\begin{alignat*}{2}\label{repRulesAA}
\hat{e}_i|\psi\rangle&\Longleftrightarrow\e{i}\psi\Ie{3},\quad  &\ii|\psi\rangle&\Longleftrightarrow I\psi\e{34},\\
\hat{e}_{ij}|\psi\rangle&\Longleftrightarrow\e{ij}\psi,\quad  &\langle\psi|\psi\rangle&\Longleftrightarrow\langle\psi^{\dagger}\psi\rangle,\\
\hat{I}\hat{e}_i|\psi\rangle&\Longleftrightarrow\Ie{i}\psi\Ie{3},\quad  & \langle\varphi|\psi\rangle&\Longleftrightarrow\langle\varphi^{\dagger}\psi\rangle-\langle\varphi^{\dagger}\psi\,\e{12}\rangle\e{12}
\end{alignat*}
where $i,j=1,2,3,4$.
The dagger operation is a combination of the reversion and spatial
inversion
\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\begin{equation*}\label{dagger}
\psi^{\dagger}=-\e{4}\tilde{\psi}\e{4}.
\end{equation*}
If applied to a general bispinor it allows to find the square of
the module
\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\begin{equation*}
\langle\psi^{\dagger}\psi\rangle=
a_0^2+a_1^2+a_2^2+a_3^2+b_0^2+b_1^2+b_2^2+b_3^2.
\end{equation*}
\end{block}
\end{column}

\end{equation}
The first term requires double application of the

I(\e{12}\psi)\e{34}=-\e{34}\psi\e{34}$. To find the eigens of the

$\inv{\psi}=-\e{4}\psi\e{4}$. With help of it
the spinor can be written as a sum of even

and its inverse  become
\begin{equation}\begin{split}
 \sH_2(\psi)&=\omega\,\e{3}\psi_{-}\e{3}-\Gamma\e{2}(\psi_{+}+\psi_{-})\e{3},\\
 \inv{\sH}_2(\psi)&=-\omega\,\e{3}\psi_{-}\e{3}-\Gamma\e{2}(\psi_{+}-\psi_{-})\e{3}.
 \end{split}\end{equation}

\subsection{Bilayer graphene}

%--29-----------------
\begin{frame}
\vspace{-20pt}
\frametitle<presentation>{\textit{Cl}$_{3,1}$ algebra case}
\begin{columns}[t]
\begin{column}{1.05\textwidth}
\begin{block}{Bilayer graphene}
In the previous example the rotors $\psi_{+}$ and $\psi_{-}$ were
decoupled what greatly simplifies the problem. Here we shall
consider more complex case of bilayer graphene (BG) where \alert{rotor equations
are coupled}.

Using matrix representation of \textit{Cl}$_{3,1}$ algebra the BG
Hamiltonian can be decomposed into sum of matrices that represent
\textit{Cl}$_{3,1}$ \setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\begin{equation}\label{hatH}
\hat{H}_{BG}=\eta(k_x\hat{e}_1+k_y\hat{e}_2)-\frac{\gamma_1}{2}\Big(\ii\,\hat{e}_{3}\hat{e}_{1}+
\hat{I}\hat{e}_2\Big)+\eta U\hat{I}\hat{e}_3.
\end{equation}
$\gamma_1$ is the interlayer coupling energy (constant). The
energy $2U$ is the potential difference between the layers.
$\eta=+1$ for $K$-valley and $\eta=-1$ for $K^{\prime}$-valley.
The electron spin is not included in this Hamiltonian. All
energies are normalized to intralayer hopping energy $\gamma_0$.
The wave vector and its components will be normalized by
$k_c=2\pi/a$, where $a$ is the graphene lattice constant. Also we
have assumed that $\hbar=1$.
\end{block}
\end{column}
\end{columns}
\end{frame}


%--30-----------------
\begin{frame}
\vspace{-20pt}
\frametitle<presentation>{\textit{Cl}$_{3,1}$ algebra case}
\begin{columns}[t]
\begin{column}{1.05\textwidth}
\begin{block}{Bilayer graphene}
Using the replacement rules the matrix Hamiltonian~\eqref{hatH} can be mapped to GA
Hamiltonian
\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\begin{equation}\label{HBG}
\sH_{BG}(\psi)=\eta\bk\psi\Ie{3}-\frac{\gamma_1}{2}
\e{2}\big(\psi-\inv{\psi}\big)\e{3}+\eta U\e{3}\psi\e{3},
\end{equation}
where $\bk=k_x\e{1}+k_y\e{2}$ is the in-plane wave vector of
electron.

To transform  the  eigenequation $\sH_{BG}(\psi)=E\psi$ to rotors
the spinor is split into even and odd parts,
$\psi=\psi_{+}+\psi_{-}$, and then the  eigenequation is
decomposed into two \alert{coupled} multivector equations for $\psi_{+}$
and $\psi_{-}$,
\begin{equation}\label{coupled}
\begin{split}
E\psi_{+}&=\eta\bk\psi_{-}\Ie{3}+\eta U\e{3}\psi_{+}\e{3},\\
E\psi_{-}&=\eta\bk\psi_{+}\Ie{3}+\eta
U\e{3}\psi_{-}\e{3}-\gamma_1\e{2}\psi_{-}\e{3}.
\end{split}
\end{equation}
The method of solution of the system~\eqref{coupled} uses the fact
that $\psi_{+}$ and $I\psi_{-}$, where $I$ is the pseudoscalar of
\textit{Cl}$_{3,1}$ algebra, are the rotors in 3D Euclidean space.
For this purpose $\psi_{-}$ is expressed  from the first equation
and inserted into the second equation of~\eqref{coupled}.
\end{block}
\end{column}
\end{columns}
\end{frame}

%--31-----------------
\begin{frame}
\vspace{-20pt}
\frametitle<presentation>{\textit{Cl}$_{3,1}$ algebra case}
\begin{columns}[t]
\begin{column}{1.05\textwidth}
\begin{block}{Bilayer graphene}
After algebraic manipulations one can construct the following
\alert{rotor equation} for $\psi_{+}$,
\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\begin{equation}\label{roteqBG}
\psi_{+}\e{3}\tilde{\psi}_{+}=NM^{-1}\equiv\hat{\ba},
\end{equation}
where
\begin{equation}
\begin{split}
M=&E[2\eta U\Ie{3}-k^{-2}\gamma_1\e{1}(k_x+k_y\e{12})^2\e{34}],\\
N=&(E^2+U^2-k^2)I+k^{-2}\eta U\gamma_1\e{1}(k_x+k_y\e{12})^2\e{4}.
\end{split}
\end{equation}
The unit vector $\hat{\ba}$ can be simplified to the following
form
\begin{equation}\label{baxi}
\begin{split}
\hat{\ba}&=\frac{\gamma_1\xi}{Ek^2}\,\bk\e{2}\bk+\frac{\eta U(2\xi+1)}{E}\,\e{3},\\
 \xi&=\frac{E^2-k^2-U^2}{4U^2+\gamma_1^2}.
 \end{split}
\end{equation}
The geometric product $\bk\e{2}\bk$ of three vectors in polar
coordinates simplifies to $\bk\e{2}\bk/k^2=\sin(2\varphi)\,\e{1}-
\cos(2\varphi)\,\e{2}$, where $\varphi$ is the angle between
$\e{1}$ and~$\bk$.
\end{block}
\end{column}
\end{columns}
\end{frame}

one can  construct the rotor for the $i$-th energy band. The
cosine of the angle $\theta^{(i)}$ between $\e{3}$ and

$\cos\theta^{(i)}=\e{3}\cdot\hat{\ba}^{(i)}$,

{\sqrt{4U^2+\gamma_1^2}\,\sqrt{(E^{(i)2}-k^2+U^2)^2+U^2\gamma_1^2}}\,.
\end{equation}

give to the rotor-spinor the following form
\begin{equation}\label{psipBG}
\psi_{+}^{(i)}=\frac{1}{\sqrt{2}}\Big(\sqrt{1+c^{(i)}}+(1-c^{(i)})\frac{\e{3}\bk\e2\bk}{k^2}\Big).
\end{equation}
After expansion one can show that $\psi_{+}^{(i)}$ consists of
scalar and two bivectors, $\e{23}$ and $\e{31}$.
\end{block}

\setlength{\abovedisplayskip}{3pt}

\section{Conclusions}
%--39----------
\begin{frame}
\vspace{-20pt}
%\frametitle<presentation>{Conclusions}
\begin{columns}[t]
\begin{column}{0.99\textwidth}
\begin{block}{Conclusions}
\begin{itemize}

\item <1-> Presence of a specific direction called the
quantization axis in the Hamiltonian can be employed to solve
quantum problems in GA.

\item <2->  Interpretation of the wave function as GA rotor can be
used  to solve quantum mechanical eigenproblems.

\item <3-> Since GA approach is coordinate-free the formulas found
in this way appear to be rather compact and may be interpreted
geometrically.

\item <4-> The method was demonstrated by a number of examples
from physics: monolayer and bilayer gaphene, two-level atoms,
quantum wells.
\end{itemize}
\end{block}
